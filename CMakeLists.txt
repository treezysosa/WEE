cmake_minimum_required(VERSION 3.22)
project(WEE VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WEE_BUILD_PLUGIN "Build JUCE instrument and standalone" ON)
enable_testing()
add_executable(wee_tests tests/CoreTests.cpp)
target_include_directories(wee_tests PRIVATE src)
add_test(NAME composition_midi COMMAND wee_tests)
add_executable(wee_layer_tests tests/LayerTests.cpp)
target_include_directories(wee_layer_tests PRIVATE src)
add_test(NAME layer_harmony_midi COMMAND wee_layer_tests)
add_executable(wee_audio_tests tests/AudioTests.cpp)
target_include_directories(wee_audio_tests PRIVATE src)
add_test(NAME dsp_flip_wav COMMAND wee_audio_tests)
add_executable(wee_schedule_tests tests/ScheduleTests.cpp)
target_include_directories(wee_schedule_tests PRIVATE src)
add_test(NAME schedule_reference COMMAND wee_schedule_tests)
if(WIN32)
 add_test(NAME installer_recovery COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/InstallerTests.ps1")
endif()
if(WEE_BUILD_PLUGIN)
 include(FetchContent)
 set(JUCE_SOURCE_DIR "" CACHE PATH "Existing JUCE 8.0.6 checkout for offline builds")
 if(JUCE_SOURCE_DIR)
  add_subdirectory("${JUCE_SOURCE_DIR}" juce)
 else()
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
 endif()
 juce_add_plugin(WEE COMPANY_NAME "WEE Audio" PLUGIN_MANUFACTURER_CODE Weea PLUGIN_CODE Wee1
  FORMATS VST3 Standalone PRODUCT_NAME "WEE" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE
  NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
 juce_add_console_app(wee_processor_tests PRODUCT_NAME "WEE Processor Tests")
 target_sources(wee_processor_tests PRIVATE tests/ProcessorTests.cpp tests/RendererTests.cpp tests/EditingTests.cpp tests/PerformanceTests.cpp)
 target_link_libraries(wee_processor_tests PRIVATE WEE juce::juce_audio_utils)
 add_test(NAME processor_audio_state COMMAND wee_processor_tests)
 set_tests_properties(processor_audio_state PROPERTIES TIMEOUT 120)
 target_sources(WEE PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp src/Export/Renderer.cpp)
 target_compile_definitions(WEE PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_link_libraries(WEE PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
endif()
